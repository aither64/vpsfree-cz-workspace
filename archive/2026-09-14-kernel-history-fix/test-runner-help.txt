NAME
    test - Run test

SYNOPSIS

    test-runner [global options] test [command options] [path-pattern]

COMMAND OPTIONS
    --cpu-overcommit=arg            - CPU overcommit factor for detected
                                      capacity (default: none)
    --cpu-reserve=arg               - CPUs to reserve from detected capacity
                                      (default: none)
    --[no-]destructive              - Determines where machine disk files are
                                      kept (default: enabled)
    -f, --[no-]fresh                - Recreate disk files
    --filter=arg                    - Filter by metadata expression (may be used
                                      more than once, default: none)
    -j, --jobs=arg                  - How many tests to run in parallel, or auto
                                      (default: 1)
    -l, --label=arg                 - Filter by label (may be used more than
                                      once, default: none)
    --max-cpus=arg                  - Maximum CPUs available to running test VMs
                                      (default: none)
    --max-memory-mib=arg            - Maximum memory available to running test
                                      VMs, in MiB (default: none)
    --max-shm-mib=arg               - Maximum /dev/shm space available to
                                      running test VMs, in MiB (default: none)
    --memory-overcommit=arg         - Memory overcommit factor for detected
                                      capacity (default: none)
    --memory-reserve-mib=arg        - Memory to reserve from detected capacity,
                                      in MiB (default: none)
    --resource-refresh-interval=arg - Seconds between detected resource capacity
                                      refreshes (default: none)
    --shm-overcommit=arg            - /dev/shm overcommit factor for detected
                                      capacity (default: none)
    --shm-reserve-mib=arg           - /dev/shm space to reserve from detected
                                      capacity, in MiB (default: none)
    --state-dir=arg                 - Directory where test logs and state are
                                      stored (default: none)
    --status-interval=arg           - Seconds between suite status messages, or
                                      0 to disable (default: none)
    --[no-]stop-on-failure          - Stop scheduling tests after the first
                                      unexpected result
    --system=arg                    - Nix system to evaluate tests for (default:
                                      x86_64-linux)
    -t, --tag=arg                   - Filter by tag (may be used more than once,
                                      default: none)
    --test-config=arg               - Path to a Nix file returning the test
                                      framework configuration (default: none)
    --timeout=arg                   - Default timeout for machine commands, in
                                      seconds (default: 900)
    -v, --[no-]verbose              - Show verbose diagnostic progress messages
